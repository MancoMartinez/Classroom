SELECT TOP 30 n.ID, n.DateCreated, n.Name, s.Value AS Status, c.Value AS Category, a.Value AS Area, n.Deleted, n.AbandonReason, n.AbandonText
FROM notify.Notification n
LEFT JOIN notify.NotificationStatus s ON s.ID = n.Status
LEFT JOIN notify.NotificationCategory c ON c.ID = n.Category
LEFT JOIN notify.NotificationArea a ON a.ID = n.Area
ORDER BY n.ID DESC;

SELECT nu.NotificationID, nu.UserID, us.Value AS UserStatus, nu.DateCreated, nu.DateUpdated, nu.[Read], nu.Deleted
FROM notify.NotificationUser nu
LEFT JOIN notify.NotificationUserStatus us ON us.ID = nu.Status
WHERE nu.NotificationID > 300
ORDER BY nu.NotificationID DESC;

SELECT TOP 30 * FROM notify.NotificationUserScheduled ORDER BY ID DESC;
